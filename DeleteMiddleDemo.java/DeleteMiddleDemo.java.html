class DeleteMiddleDemo {
    int data;
    Node next;

    Node(int data) {
        this.data = data;
        this.next = null;
    }
}

class SinglyLinkedList {
    Node head;

    // Add node to the end of the list
    public void add(int data) {
        Node newNode = new Node(data);
        if (head == null) {
            head = newNode;
            return;
        }
        Node current = head;
        while (current.next != null) {
            current = current.next;
        }
        current.next = newNode;
    }


    public void display() {
        Node current = head;
        while (current != null) {
            System.out.print(current.data + " --> ");
            current = current.next;
        }
        System.out.println("NULL");
    }


    public void deleteMiddle() {
        if (head == null) {
            return;
        }


        if (head.next == null) {
            head = null;
            return;
        }

        Node slow = head;
        Node fast = head;
        Node prev = null;


        while (fast != null && fast.next != null) {
            fast = fast.next.next;
            prev = slow;
            slow = slow.next;
        }


        if (prev != null) {
            prev.next = slow.next;
        }
    }
}

public class DeleteMiddleDemo {

    public static void main(String[] args) {

        System.out.println("=== Delete Middle Node Demo ===");
        System.out.println();



        SinglyLinkedList list1 = new SinglyLinkedList();
        list1.add(1);
        list1.add(2);
        list1.add(3);
        list1.add(4);
        list1.add(5);

        System.out.println("Original List 1:");
        list1.display();

        list1.deleteMiddle();

        System.out.println("After deleting middle:");
        list1.display();

        System.out.println();


        SinglyLinkedList list2 = new SinglyLinkedList();
        list2.add(10);
        list2.add(20);
        list2.add(30);
        list2.add(40);
        list2.add(50);
        list2.add(60);

        System.out.println("Original List 2:");
        list2.display();

        list2.deleteMiddle();

        System.out.println("After deleting middle:");
        list2.display();

        System.out.println();


        SinglyLinkedList list3 = new SinglyLinkedList();
        list3.add(999);

        System.out.println("Original List 3:");
        list3.display();

        list3.deleteMiddle();

        System.out.println("After deleting middle:");
        list3.display();
    }
}
