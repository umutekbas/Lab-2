public class SinglyLinkedList {


    private Node head;



    public SinglyLinkedList() {
        head = null;
    }


    public void add(int data) {
        Node newNode = new Node(data);

        // liste bossa direkt head yap
        if (head == null) {
            head = newNode;
        } else {
            // son elemana kadar git
            Node temp = head;
            while (temp.next != null) {
                temp = temp.next;
            }
            temp.next = newNode;
        }
    }



    public void display() {
        Node temp = head;
        while (temp != null) {
            System.out.print(temp.data + " --> ");
            temp = temp.next;
        }
        System.out.println("NULL");
    }



    public Node findMiddle() {
        if (head == null) {
            return null;
        }


        int count = 0;
        Node temp = head;
        while (temp != null) {
            count++;
            temp = temp.next;
        }


        int mid = count / 2;


        temp = head;
        for (int i = 0; i < mid; i++) {
            temp = temp.next;
        }

        return temp;
    }


    public Node getHead() {
        return head;
    }
}
